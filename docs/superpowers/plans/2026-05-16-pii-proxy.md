# PII Proxy Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在 claudecodeui 中加入一个 FastAPI PII 脱敏代理，拦截发往 Anthropic API 的请求，用 AES 加密替换 PII，并在非流式响应中还原。

**Architecture:** Python FastAPI 进程与 Node 服务并列启动（通过 `concurrently`），监听 `127.0.0.1:18090`。Node server（`claude-sdk.js`）在 `PII_PROXY_PORT` 环境变量存在时，以与 tap 相同的 `extraArgs.settings.env.ANTHROPIC_BASE_URL` 方式将请求路由到代理。代理使用 Presidio `AnalyzerEngine` 检测实体，自定义 `<PII:TYPE>AES_CBC</PII>` 标签包裹加密值，响应时正则匹配解密还原。

**Tech Stack:** Python 3.10+, FastAPI, uvicorn, httpx, presidio-analyzer, cryptography (spaCy en_core_web_lg), Node.js ESM

- [ ] **Step 1: 编写 pii_proxy.py**

```python
import os
import re
import json
import base64
import secrets
import logging
from typing import Optional

import httpx
from fastapi import FastAPI, Request, Response
from fastapi.responses import StreamingResponse
from cryptography.hazmat.primitives.ciphers import Cipher, algorithms, modes
from cryptography.hazmat.primitives import padding as crypto_padding
from cryptography.hazmat.backends import default_backend
from presidio_analyzer import AnalyzerEngine

logging.basicConfig(level=logging.INFO, format="%(levelname)s %(name)s: %(message)s")
logger = logging.getLogger("pii-proxy")

# ---------------------------------------------------------------------------
# Config
# ---------------------------------------------------------------------------

UPSTREAM_URL = os.getenv("PII_PROXY_UPSTREAM_URL", "https://api.anthropic.com").rstrip("/")
ENTITY_TYPES = [
    e.strip()
    for e in os.getenv(
        "PII_PROXY_ENTITIES",
        "PERSON,PHONE_NUMBER,EMAIL_ADDRESS,CREDIT_CARD,LOCATION",
    ).split(",")
]

# AES-256 key stored as 64-char hex; auto-generate if absent
_key_hex = os.getenv("PII_PROXY_AES_KEY", "")
if len(_key_hex) == 64:
    AES_KEY = bytes.fromhex(_key_hex)
else:
    AES_KEY = secrets.token_bytes(32)
    logger.info("Generated AES key (set PII_PROXY_AES_KEY=%s to persist)", AES_KEY.hex())

# ---------------------------------------------------------------------------
# AES-CBC helpers
# ---------------------------------------------------------------------------

def _encrypt(plaintext: str) -> str:
    iv = secrets.token_bytes(16)
    padder = crypto_padding.PKCS7(128).padder()
    padded = padder.update(plaintext.encode()) + padder.finalize()
    cipher = Cipher(algorithms.AES(AES_KEY), modes.CBC(iv), backend=default_backend())
    enc = cipher.encryptor()
    ct = enc.update(padded) + enc.finalize()
    return base64.b64encode(iv + ct).decode()


def _decrypt(b64data: str) -> str:
    raw = base64.b64decode(b64data)
    iv, ct = raw[:16], raw[16:]
    cipher = Cipher(algorithms.AES(AES_KEY), modes.CBC(iv), backend=default_backend())
    dec = cipher.decryptor()
    padded = dec.update(ct) + dec.finalize()
    unpadder = crypto_padding.PKCS7(128).unpadder()
    return (unpadder.update(padded) + unpadder.finalize()).decode()

# ---------------------------------------------------------------------------
# PII anonymize / deanonymize
# ---------------------------------------------------------------------------

_TAG_RE = re.compile(r"<PII:[A-Z_]+>[^<]+</PII>")

_analyzer = AnalyzerEngine()


def anonymize_text(text: str) -> str:
    if not text or not text.strip():
        return text
    results = _analyzer.analyze(text=text, language="en", entities=ENTITY_TYPES)
    if not results:
        return text
    # Replace right-to-left so offsets stay valid
    results.sort(key=lambda r: r.start, reverse=True)
    chars = list(text)
    for r in results:
        encrypted = _encrypt(text[r.start : r.end])
        tag = f"<PII:{r.entity_type}>{encrypted}</PII>"
        chars[r.start : r.end] = list(tag)
    return "".join(chars)


def deanonymize_text(text: str) -> str:
    def _replace(m: re.Match) -> str:
        inner = m.group(0)
        # Extract base64 payload between > and </PII>
        payload = inner.split(">", 1)[1].rsplit("</PII>", 1)[0]
        try:
            return _decrypt(payload)
        except Exception:
            return inner  # leave as-is if decrypt fails

    return _TAG_RE.sub(_replace, text)


def _anonymize_messages(messages: list) -> list:
    out = []
    for msg in messages:
        msg = dict(msg)
        content = msg.get("content")
        if isinstance(content, str):
            msg["content"] = anonymize_text(content)
        elif isinstance(content, list):
            new_blocks = []
            for block in content:
                block = dict(block)
                if block.get("type") == "text" and isinstance(block.get("text"), str):
                    block["text"] = anonymize_text(block["text"])
                new_blocks.append(block)
            msg["content"] = new_blocks
        out.append(msg)
    return out


def _deanonymize_obj(obj):
    if isinstance(obj, str):
        return deanonymize_text(obj)
    if isinstance(obj, dict):
        return {k: _deanonymize_obj(v) for k, v in obj.items()}
    if isinstance(obj, list):
        return [_deanonymize_obj(item) for item in obj]
    return obj

# ---------------------------------------------------------------------------
# FastAPI app
# ---------------------------------------------------------------------------

app = FastAPI(title="PII Proxy")
_client = httpx.AsyncClient(timeout=120.0, follow_redirects=True)

_HOP_BY_HOP = {"host", "content-length", "transfer-encoding", "connection", "keep-alive"}


@app.get("/health")
async def health():
    return {"status": "ok", "upstream": UPSTREAM_URL}


@app.api_route("/{path:path}", methods=["GET", "POST", "PUT", "DELETE", "PATCH"])
async def proxy(request: Request, path: str):
    body = await request.body()

    # Anonymize PII in POST /v1/messages
    if request.method == "POST" and path == "v1/messages":
        try:
            data = json.loads(body)
            if "messages" in data:
                data["messages"] = _anonymize_messages(data["messages"])
                body = json.dumps(data).encode()
        except Exception as exc:
            logger.warning("Failed to anonymize request: %s", exc)

    # Forward to upstream
    fwd_headers = {k: v for k, v in request.headers.items() if k.lower() not in _HOP_BY_HOP}
    upstream_url = f"{UPSTREAM_URL}/{path}"
    if request.url.query:
        upstream_url += f"?{request.url.query}"

    upstream_resp = await _client.request(
        method=request.method,
        url=upstream_url,
        headers=fwd_headers,
        content=body,
    )

    content_type = upstream_resp.headers.get("content-type", "")
    resp_headers = {k: v for k, v in upstream_resp.headers.items() if k.lower() not in _HOP_BY_HOP}

    # SSE: pass through directly
    if "text/event-stream" in content_type:
        async def _stream():
            async for chunk in upstream_resp.aiter_bytes():
                yield chunk
        return StreamingResponse(_stream(), status_code=upstream_resp.status_code, headers=resp_headers)

    # Non-streaming: deanonymize response
    resp_body = upstream_resp.content
    if "application/json" in content_type:
        try:
            resp_data = json.loads(resp_body)
            resp_data = _deanonymize_obj(resp_data)
            resp_body = json.dumps(resp_data).encode()
        except Exception as exc:
            logger.warning("Failed to deanonymize response: %s", exc)

    resp_headers["content-length"] = str(len(resp_body))
    return Response(
        content=resp_body,
        status_code=upstream_resp.status_code,
        headers=resp_headers,
        media_type=content_type,
    )
```
